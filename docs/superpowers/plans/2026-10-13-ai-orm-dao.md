```sql
CREATE TABLE ai_knowledge_base
(

    name             VARCHAR(50)  NOT NULL COMMENT '知识库名称',
    file_count       INT          NOT NULL DEFAULT 0 COMMENT '文档数量',

--   file_chunk_size/overlap/split_type=chunking 参数的文件级快照，ingestion 编排时写入）
CREATE TABLE ai_knowledge_base_files
(
    id                 BIGINT       NOT NULL AUTO_INCREMENT COMMENT '文件ID（主键）',
    kb_id              BIGINT       NOT NULL COMMENT '知识库ID',
    file_path          VARCHAR(512) NOT NULL COMMENT '文档地址',
    file_name          VARCHAR(50)  NOT NULL COMMENT '文档名称',
    file_chunk_count   INT          NOT NULL COMMENT '文档分片数',
    file_chunk_size    INT          NOT NULL COMMENT '分片大小',
    file_chunk_overlap INT          NOT NULL COMMENT '分片重叠大小',
    file_split_type    VARCHAR(50)  NOT NULL COMMENT '分片类型：parent-child（父子分片）/ recursion（递归分片）',
    description        VARCHAR(512) NULL COMMENT '描述信息',
    create_by          VARCHAR(30)  NULL COMMENT '创建人',
    create_time        DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_by          VARCHAR(30)  NULL COMMENT '更新人',
    update_time        DATETIME     NULL COMMENT '更新时间',
    deleted            TINYINT      NOT NULL DEFAULT 0 COMMENT '逻辑删除：0正常 1已删',
    PRIMARY KEY (id),
    UNIQUE KEY uk_kb_file_path (kb_id, file_path),
    -- ON 不可用时替换上一行为：UNIQUE KEY uk_kb_file_path (kb_id, file_path(191)),

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci COMMENT='AI知识库文件';
```

```python
"""临时 SQL runner（验后删）：执行 SQL 剧本文件（UTF-8，按分号拆句，幂等语句安全重放）。"""

import sys

import pymysql

CONN = dict(host="127.0.0.1", user="root", password="", charset="utf8mb4")


def run(path: str) -> None:
    with open(path, encoding="utf-8") as f:
        raw = f.read()
    stmts = [s.strip() for s in raw.split(";") if s.strip()]
    conn = pymysql.connect(**CONN)
    try:
        with conn.cursor() as cur:
            for i, s in enumerate(stmts, 1):
                cur.execute(s)
                print(f"[{i}/{len(stmts)}] ok: {s.splitlines()[0][:60]}")
        conn.commit()
    finally:
        conn.close()


if __name__ == "__main__":
    run(sys.argv[1])
```

```python
"""DB 连接基建：engine/session 进程内单例工厂与优雅关停（spec D6/D7）。

- 惰性首连：engine 创建零网络，连接在首个真实操作时建立——DB 不可达表现为请求期
  sqlalchemy 异常（全局 handler 转 R 500），不阻断启动（对齐注册降级哲学）；
- charset=utf8mb4 显式：Windows 连接层中文乱码防线（CLAUDE.md 记档）；
- pool_recycle：防 MySQL wait_timeout（5.7 默认 8h）stale 连接；
- pool_pre_ping 不开（每请求多一次 ping，本机量级 recycle 足够，spec D6 记档）。
"""

import logging
from functools import lru_cache
from urllib.parse import quote_plus

from sqlalchemy.ext.asyncio import (
    AsyncEngine,
    AsyncSession,
    async_sessionmaker,
    create_async_engine,
)

from cloud_base_ai.config import DatabaseSettings, get_settings

logger = logging.getLogger(__name__)


def build_url(cfg: DatabaseSettings) -> str:
    """组装 aiomysql 连接 URL（user/password 经 quote_plus 防御特殊字符）。"""
    auth = f"{quote_plus(cfg.user)}:{quote_plus(cfg.password)}"
    return f"mysql+aiomysql://{auth}@{cfg.host}:{cfg.port}/{cfg.database}?charset=utf8mb4"


@lru_cache
def get_engine() -> AsyncEngine:
    """async engine 进程内单例（lru_cache 工厂，对齐 model_factory 口径）。"""
    cfg = get_settings().db
    return create_async_engine(
        build_url(cfg),
        pool_size=cfg.pool_size,
        max_overflow=cfg.max_overflow,
        pool_recycle=cfg.pool_recycle,
        echo=cfg.echo,
    )


@lru_cache
def get_session_factory() -> async_sessionmaker[AsyncSession]:
    """session 工厂单例；expire_on_commit=False（async 防隐式 IO 硬要求）。"""
    return async_sessionmaker(get_engine(), expire_on_commit=False)


async def dispose_engine() -> None:
    """优雅关停连接池（lifespan shutdown 调用；幂等，dispose 后清缓存防复用已关引擎）。"""
    try:
        await get_engine().dispose()
        logger.debug("db engine disposed")
    finally:
        get_engine.cache_clear()
        get_session_factory.cache_clear()
```

```python
"""SQLite 内存测试替身：每测试全新引擎建表（StaticPool 单连接共享内存库，spec D9）。

方言差异记档：SQLite 覆盖不到 MySQL 特有面（charset/时区/错误细节），uk 撞重以
sqlalchemy.exc.IntegrityError 统一断言；MySQL 特有面由真库冒烟（smoke_db.py）覆盖。

from collections.abc import AsyncIterator
from contextlib import asynccontextmanager

from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker, create_async_engine
from sqlalchemy.pool import StaticPool

from cloud_base_ai.db.base import Base


@asynccontextmanager
async def make_db_session() -> AsyncIterator[AsyncSession]:
    """yield 一个全新内存库上的 session（结束销毁引擎，无跨测试泄漏）。"""
    engine = create_async_engine("sqlite+aiosqlite://", poolclass=StaticPool)
    try:
        async with engine.begin() as conn:
            await conn.run_sync(Base.metadata.create_all)
        factory = async_sessionmaker(engine, expire_on_commit=False)
        async with factory() as session:
            yield session
    finally:
        await engine.dispose()
```

```python
"""RAG 域 DAO（数据操作对象）：知识库/知识库文件两套动词集（spec D4/D5）。

口径：
- 模块函数收 session：写方法以 flush 收尾、永不 commit——事务边界归编排层
  （service 的 async with session.begin() 块，Java @Transactional 对应物）；
- 软删：所有读路径显式 filter(deleted == 0)；delete = UPDATE SET deleted=1 + 审计两值；
- 审计：save/update 走 fill_*_audit 公共路径（account 来源 contextvar）；
- 异常零转译：IntegrityError（uk 撞重/墓碑占键）等原样上抛，兜底转译在 service（端点轮）；
- 分页 clamp：page>=1、1<=size<=MAX_PAGE_SIZE（对齐 Java 分页插件 maxLimit 200），排序 id DESC。
"""

from collections.abc import Sequence
from datetime import datetime

from sqlalchemy import func, select, update
from sqlalchemy.ext.asyncio import AsyncSession

from cloud_base_ai.db.base import (
    SoftDeleteMixin,
    current_account,
    fill_create_audit,
    fill_update_audit,
)
from cloud_base_ai.rag.entities import AiKnowledgeBase, AiKnowledgeBaseFiles

MAX_PAGE_SIZE = 200


def _clamp_page(page: int, size: int) -> tuple[int, int]:
    """分页参数钳制：page 至少 1、size 至少 1 至多 MAX_PAGE_SIZE。"""
    return max(1, page), max(1, min(size, MAX_PAGE_SIZE))


# ---------- AiKnowledgeBase ----------


async def save_kb(session: AsyncSession, kb: AiKnowledgeBase) -> AiKnowledgeBase:
    """新增知识库：审计四值 → add+flush（id 回填）→ refresh（回填 server_default 生成值——
    async 下 flush 后直接访问 server 生成列会隐式 SELECT 报 MissingGreenlet，refresh 是显式合法路径）。"""
    fill_create_audit(kb)
    session.add(kb)
    await session.flush()
    await session.refresh(kb)
    return kb


async def find_kb_by_id(session: AsyncSession, kb_id: int) -> AiKnowledgeBase | None:
    """按主键查活记录（软删不可见）。"""
    stmt = select(AiKnowledgeBase).where(
        AiKnowledgeBase.id == kb_id,
        AiKnowledgeBase.deleted == SoftDeleteMixin.DeletedEnum.NORMAL,
    )
    return (await session.execute(stmt)).scalar_one_or_none()


async def find_kb_by_name(session: AsyncSession, name: str) -> AiKnowledgeBase | None:
    """按名称查活记录（创建/改名查重；墓碑占键由 uk_name 的 IntegrityError 兜底）。"""
    stmt = select(AiKnowledgeBase).where(
        AiKnowledgeBase.name == name,
        AiKnowledgeBase.deleted == SoftDeleteMixin.DeletedEnum.NORMAL,
    )
    return (await session.execute(stmt)).scalar_one_or_none()


async def update_kb(session: AsyncSession, kb: AiKnowledgeBase) -> AiKnowledgeBase:
    """修改知识库：审计两值 → merge → flush（消费方典型流：find 拿活实体→改字段→本方法）。"""
    fill_update_audit(kb)
    merged = await session.merge(kb)
    await session.flush()
    return merged


async def delete_kb_by_id(session: AsyncSession, kb_id: int) -> int:
    """软删知识库：UPDATE SET deleted=1 + 审计两值；返回影响行数（二次删为 0）。"""
    stmt = (
        update(AiKnowledgeBase)
        .where(
            AiKnowledgeBase.id == kb_id,
            AiKnowledgeBase.deleted == SoftDeleteMixin.DeletedEnum.NORMAL,
        )
        .values(
            deleted=SoftDeleteMixin.DeletedEnum.DELETED,
            update_by=current_account(),
            update_time=datetime.now(),
        )
        .execution_options(synchronize_session=False)
    )
    result = await session.execute(stmt)
    return result.rowcount


async def page_list_kbs(
    session: AsyncSession, page: int, size: int, name: str | None = None
) -> tuple[Sequence[AiKnowledgeBase], int]:
    """分页查知识库（name 模糊过滤可选；total 与 rows 同条件；排序 id DESC 主键反向扫）。"""
    page, size = _clamp_page(page, size)
    conds = [AiKnowledgeBase.deleted == SoftDeleteMixin.DeletedEnum.NORMAL]
    if name:
        conds.append(AiKnowledgeBase.name.like(f"%{name}%"))
    total = (
        await session.execute(select(func.count()).select_from(AiKnowledgeBase).where(*conds))
    ).scalar_one()
    rows = (
        await session.execute(
            select(AiKnowledgeBase)
            .where(*conds)
            .order_by(AiKnowledgeBase.id.desc())
            .limit(size)
            .offset((page - 1) * size)
        )
    ).scalars().all()
    return rows, total


async def list_kbs(session: AsyncSession) -> Sequence[AiKnowledgeBase]:
    """全量活知识库（id 升序；管理面下拉等小量场景）。"""
    stmt = (
        select(AiKnowledgeBase)
        .where(AiKnowledgeBase.deleted == SoftDeleteMixin.DeletedEnum.NORMAL)
        .order_by(AiKnowledgeBase.id.asc())
    )
    return (await session.execute(stmt)).scalars().all()


async def count_kbs(session: AsyncSession) -> int:
    """活知识库计数。"""
    stmt = (
        select(func.count())
        .select_from(AiKnowledgeBase)
        .where(AiKnowledgeBase.deleted == SoftDeleteMixin.DeletedEnum.NORMAL)
    )
    return (await session.execute(stmt)).scalar_one()
```

```python
"""真库冒烟（一次性，验后删）：aiomysql×SQLAlchemy×MySQL5.7×Windows 组合实证 + DDL 修订实证。

数据 e2e 前缀 + 结束物理清理（规范第 6 条口子）。运行：cd cloud-base-ai && uv run python smoke_db.py
"""

import asyncio
import sys

if sys.platform == "win32":
    asyncio.set_event_loop_policy(asyncio.WindowsSelectorEventLoopPolicy())

from sqlalchemy import delete, text
from sqlalchemy.exc import IntegrityError

from cloud_base_ai.db import dispose_engine, get_session_factory
from cloud_base_ai.rag import dao
from cloud_base_ai.rag.entities import AiKnowledgeBase, AiKnowledgeBaseFiles

NAME = "e2e_冒烟知识库_中文"


async def main() -> None:
    checks: list[tuple[str, bool]] = []
    factory = get_session_factory()
    try:
        async with factory() as s:
            # 1. 建连（组合实证：aiomysql 驱动层）
            ok = (await s.execute(text("SELECT 1"))).scalar() == 1
            checks.append(("1 建连 SELECT 1", ok))

            # 2. 中文 roundtrip（charset=utf8mb4 实证）
            async with s.begin():
                kb = await dao.save_kb(s, AiKnowledgeBase(name=NAME, description="冒烟"))
            found = await dao.find_kb_by_id(s, kb.id)
            checks.append(("2 中文 roundtrip", found is not None and found.name == NAME))

            # 3. update + page_list
            found.description = "冒烟2"
            async with s.begin():
                await dao.update_kb(s, found)
            rows, total = await dao.page_list_kbs(s, 1, 10, name="e2e_冒烟")
            checks.append(("3 update+page_list", total >= 1 and rows[0].description == "冒烟2"))

            # 4. 软删过滤 + 墓碑占键（uk_name 实证）
            async with s.begin():
                n = await dao.delete_kb_by_id(s, kb.id)
            gone = (await dao.find_kb_by_id(s, kb.id)) is None
            tombstone_dup = False
            try:
                async with s.begin():
                    await dao.save_kb(s, AiKnowledgeBase(name=NAME))
            except IntegrityError:
                tombstone_dup = True
            checks.append(("4 软删+墓碑占键 uk_name", n == 1 and gone and tombstone_dup))

            # 5. files 套件（uk_kb_file_path 实证）
            async with s.begin():
                kb2 = await dao.save_kb(s, AiKnowledgeBase(name="e2e_冒烟库2"))
            async with s.begin():
                f = await dao.save_kb_file(s, AiKnowledgeBaseFiles(
                    kb_id=kb2.id, file_path="/data/e2e.md", file_name="e2e.md",
                    file_chunk_count=2, file_chunk_size=500, file_chunk_overlap=50,
                    file_split_type="recursion"))
            dup = False
            try:
                async with s.begin():
                    await dao.save_kb_file(s, AiKnowledgeBaseFiles(
                        kb_id=kb2.id, file_path="/data/e2e.md", file_name="e2e2.md",
                        file_chunk_count=1, file_chunk_size=500, file_chunk_overlap=50,
                        file_split_type="recursion"))
            except IntegrityError:
                dup = True
            cnt = await dao.count_kb_files(s, kb2.id)
            checks.append(("5 files uk 撞重", dup and cnt == 1 and f.id is not None))

            # 6. DDL 修订实证
            ddl = (await s.execute(text("SHOW CREATE TABLE ai_knowledge_base"))).mappings().first()
            checks.append(("6 uk_name 物理存在", ddl is not None and "uk_name" in ddl["Create Table"]))
    finally:

        await dispose_engine()

    for label, ok in checks:
        print(("PASS " if ok else "FAIL ") + label)
    sys.exit(0 if all(ok for _, ok in checks) else 1)


asyncio.run(main())
```
